def crear_usuario(
    connection,
    nombre: str,
    email: str,
    contrasenia: str,
    telefono: str,
) -> str:
    """Crear Usuario (cu)"""
    cursor = connection.cursor()
    try:
        connection.start_transaction()
        query = "INSERT INTO usuario(nombre, email, contrasenia, telefono) VALUES (%s, %s, %s, %s)"
        cursor.execute(query, (nombre, email, contrasenia, telefono))
        connection.commit()
        return "Usuario creado con éxito."
    except Error:
        connection.rollback()
        return "Ocurrio un error."
    finally:
        cursor.close()
